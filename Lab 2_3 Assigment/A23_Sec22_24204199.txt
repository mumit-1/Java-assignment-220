public static void sumOddAppend(Node dh) {
    Node x = dh;
    int sum = 0;
    while (x.next!=dh){
        int need =(int)x.next.elem;
        if (need%2!=0){
            sum+=need;
            x.next=x.next.next;
        }
        else{
            x=x.next;
        }
    }
    Node x2 = new Node(sum);
    x.next=x2;
    x2.next=dh;

}

            }

        }
        
    }
