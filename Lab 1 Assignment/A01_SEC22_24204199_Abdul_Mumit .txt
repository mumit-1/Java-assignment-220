                    if((i+1)>=0 && (i+1)<arena.length){
                        if(arena[i+1][j]==2) n++;
                    }
                    if((j+1)>=0 && (j+1)<arena[i].length){
                       if(arena[i][j+1]==2) n++;

                        if(arena[i][j-1]==2) n++;

                    }
