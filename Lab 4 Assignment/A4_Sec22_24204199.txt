        int need = hashFunction(key);
        PairNode head = ht[need];
        if(head==null) return;
        PairNode n = null;
        while(head!=null){
            if(head.key==key){
                if(n==null){
                     ht[need]=head.next;   
                }
               else{
                n.next=head.next;
               }
               return;
            }
            n=head;
            head=head.next;
        }
